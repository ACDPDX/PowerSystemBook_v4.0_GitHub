+   PARAMS:
+    UN_verkettet=20000
+    IN=14.434k
+    Pel=400e6  ;UN_verkettet*IN*sqrt(3)*cosphi of 0.8
+    fn=50
+    Kc=0.5
+    UE0=600V ; Power of exciter = 0.25-5% * SN machine (smaller values for higher SN)
+    IE0=2000A
+    p=2
+    Streuung=0.05 ;
+    rcu=0.001

+   err0=0

+   n0=1500
+   n1={n0} ;starts with 1500U/min
+   alpha_0=0

.PARAM
+   a1={a};  Saturation parameter d-axis
+   m1={m};  Saturation parameter d-axis
+   a2={a1}; Saturation parameter q-axis
+   m2={m1}; Saturation parameter q-axis
+   UN={UN_verkettet/sqrt(3)}
+   ZN={UN/IN}
+   Rs={ZN*rcu}
+   omega_el={(2*pi)*fN}
+   Ls={Streuung*ZN/Kc/omega_el}
+   PsiN={sqrt(2)*UN/omega_el}
+   RE={UE0/IE0}
+   RD={Rs/Daempferfaktor}
+   RDx={RD} ; can be overwritten by setting this parameters on modelline
+   RDy={RD} ; can be overwritten by setting this parameters on modelline
+   LDx={Ls} ; can be overwritten by setting this parameters on modelline
+   LDy={Ls} ; can be overwritten by setting this parameters on modelline
+   LE={Ls} ; can be overwritten by setting this parameters on modelline
+   R0={Rs} ; can be overwritten by setting this parameters on modelline
+   L0={Ls} ; can be overwritten by setting this parameters on modelline (X0 is 0.03-0.1 p.u)
+   Lsd={Ls} 
+   Lsq={Ls}
+   XD={UN/IN/Kc}
+   XQ={XD*Schenkelpolfaktor}
+   g={UE0/(UN*sqrt(2)*RE/(XD-omega_el*Lsd))} ; updated formula for g parameter
+   g1={UE0/(UN*sqrt(2)*RE/(XQ-omega_el*Lsq))} ; updated formula for salient pole machine
+   ImNx={IE0/g} ; d-axis magnetizing current
+   ImNy={IE0/g1} ;q-axis magnetizing current
+   MN={Pel*(1-rbg-rcu)*p/omega_el}

Eimx    imxa1  0  VALUE={ImNx*(a1*i(vphx)/PsiN+(1-a1)*PWRS(i(vphx)/PsiN,m1))}

Gpsiy    0  isy  VALUE={v(psisy)}

Vn n 0 {n1}
